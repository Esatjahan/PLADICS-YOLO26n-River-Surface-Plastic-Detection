from ultralytics import YOLO
from ensemble_boxes import weighted_boxes_fusion
import cv2
import os
import yaml
from pathlib import Path

model_paths = [
    "D:/PLADICS/experiments_clean_final/yolo26n_clean_100/weights/best.pt",
    "D:/PLADICS/experiments_clean_final/yolo11n_clean/weights/best.pt"
]

test_images_dir = "D:/PLADICS/datasets_clean/merged_clean_dataset/images/test"
output_dir = "D:/PLADICS/experiments_ensemble/yolo26n_yolo11n_wbf_predictions"
os.makedirs(output_dir, exist_ok=True)

models = [YOLO(p) for p in model_paths]

conf_thres = 0.25
iou_thr = 0.55
skip_box_thr = 0.001
imgsz = 640

for img_name in os.listdir(test_images_dir):
    if not img_name.lower().endswith((".jpg", ".jpeg", ".png")):
        continue

    img_path = os.path.join(test_images_dir, img_name)
    img = cv2.imread(img_path)

    if img is None:
        print("Could not read:", img_path)
        continue

    h, w = img.shape[:2]

    boxes_list = []
    scores_list = []
    labels_list = []

    for model in models:
        result = model(img_path, imgsz=imgsz, conf=conf_thres, verbose=False)[0]

        boxes = []
        scores = []
        labels = []

        for box in result.boxes:
            x1, y1, x2, y2 = box.xyxy[0].cpu().numpy()
            conf = float(box.conf[0].cpu().numpy())
            cls = int(box.cls[0].cpu().numpy())

            # Normalize boxes for WBF
            boxes.append([
                max(0, x1 / w),
                max(0, y1 / h),
                min(1, x2 / w),
                min(1, y2 / h)
            ])
            scores.append(conf)
            labels.append(cls)

        boxes_list.append(boxes)
        scores_list.append(scores)
        labels_list.append(labels)

    if len(boxes_list[0]) == 0 and len(boxes_list[1]) == 0:
        cv2.imwrite(os.path.join(output_dir, img_name), img)
        continue

    boxes, scores, labels = weighted_boxes_fusion(
        boxes_list,
        scores_list,
        labels_list,
        weights=[1, 1],
        iou_thr=iou_thr,
        skip_box_thr=skip_box_thr
    )

    for box, score, label in zip(boxes, scores, labels):
        x1 = int(box[0] * w)
        y1 = int(box[1] * h)
        x2 = int(box[2] * w)
        y2 = int(box[3] * h)

        cv2.rectangle(img, (x1, y1), (x2, y2), (0, 255, 0), 2)
        cv2.putText(
            img,
            f"plastic {score:.2f}",
            (x1, max(20, y1 - 5)),
            cv2.FONT_HERSHEY_SIMPLEX,
            0.6,
            (0, 255, 0),
            2
        )

    cv2.imwrite(os.path.join(output_dir, img_name), img)

print("WBF ensemble prediction complete.")

from ultralytics import YOLO
import cv2
from collections import defaultdict

model_path = "D:/PLADICS/experiments_clean_final/yolo26n_clean_100/weights/best.pt"
model = YOLO(model_path)

source = 0
cap = cv2.VideoCapture(source)

conf_thres = 0.25
imgsz = 640

track_history = defaultdict(int)
min_stable_frames = 5

alpha = 0.5   # bbox area weight
beta = 0.3    # confidence weight
gamma = 0.2   # center alignment weight

while True:
    ret, frame = cap.read()
    if not ret:
        break

    h, w = frame.shape[:2]

    results = model.track(
        frame,
        imgsz=imgsz,
        conf=conf_thres,
        tracker="bytetrack.yaml",
        persist=True,
        verbose=False
    )[0]

    candidates = []

    if results.boxes is not None and results.boxes.id is not None:
        for box, track_id in zip(results.boxes, results.boxes.id):
            x1, y1, x2, y2 = box.xyxy[0].cpu().numpy()
            conf = float(box.conf[0].cpu().numpy())
            tid = int(track_id.cpu().numpy())

            track_history[tid] += 1

            bw = x2 - x1
            bh = y2 - y1
            area = (bw * bh) / (w * h)

            cx = (x1 + x2) / 2
            center_alignment = 1 - abs(cx - (w / 2)) / (w / 2)

            priority = alpha * area + beta * conf + gamma * center_alignment

            if track_history[tid] >= min_stable_frames:
                candidates.append({
                    "track_id": tid,
                    "box": [x1, y1, x2, y2],
                    "conf": conf,
                    "area": area,
                    "cx": cx,
                    "priority": priority
                })

    if len(candidates) == 0:
        action = "SEARCH"
        target = None
    else:
        target = max(candidates, key=lambda x: x["priority"])
        cx = target["cx"]
        area = target["area"]

        if cx < w / 3:
            action = "MOVE LEFT"
        elif cx > 2 * w / 3:
            action = "MOVE RIGHT"
        else:
            action = "MOVE FORWARD"

        if area > 0.20:
            action = "STOP AND COLLECT"

    if target is not None:
        x1, y1, x2, y2 = target["box"]
        tid = target["track_id"]

        cv2.rectangle(frame, (int(x1), int(y1)), (int(x2), int(y2)), (0, 255, 0), 2)
        cv2.putText(frame, f"ID:{tid} P:{target['priority']:.2f}", (int(x1), int(y1)-10),
                    cv2.FONT_HERSHEY_SIMPLEX, 0.6, (0, 255, 0), 2)

    cv2.putText(frame, f"Action: {action}", (20, 40),
                cv2.FONT_HERSHEY_SIMPLEX, 1.0, (0, 0, 255), 2)

    cv2.imshow("YOLO26n + ByteTrack + Priority Decision", frame)

    if cv2.waitKey(1) & 0xFF == ord("q"):
        break

cap.release()
cv2.destroyAllWindows()
